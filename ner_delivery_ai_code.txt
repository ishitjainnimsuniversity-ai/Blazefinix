        function generateCaptcha() {
            const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
            let code = '';
            for (let i = 0; i < 5; i++) code += chars.charAt(Math.floor(Math.random() * chars.length));
            generatedCaptchaCode = code;
            document.getElementById('captcha-display').innerText = code.split('').join(' ');
        }

        function verifyCustomerCaptcha() {
            const waybillRaw = document.getElementById('cust-track-input').value.trim().toUpperCase();
            const userCaptcha = document.getElementById('captcha-user-input').value.trim().toUpperCase();

            if (!waybillRaw) { alert('Please enter a valid Tracking ID.'); return; }
            if (userCaptcha !== generatedCaptchaCode) {
                alert('⚠️ Invalid Captcha code!');
                generateCaptcha();
                document.getElementById('captcha-user-input').value = '';
                return;
            }

            const custInfo = CUSTOMER_DATABASE[waybillRaw] || CUSTOMER_DATABASE['NER-804'];
            document.getElementById('customer-name-heading').innerText = custInfo.name;
            document.getElementById('customer-waybill-label').innerText = custInfo.waybill;
            document.getElementById('customer-driver-name').innerText = custInfo.driver;
            document.getElementById('customer-driver-vehicle').innerText = '4x4 Logistics Vehicle #' + custInfo.vehicle;

            closeCustomerAuthModal();
            activateRoleView('customer');
        }

        function switchRoute(key) {
            currentRouteKey = key;

            document.getElementById('map-corridor-title').innerText = route.name;
            document.getElementById('rain-percent-badge').innerText = route.rainProbability + '%';
            document.getElementById('landslide-percent-badge').innerText = route.landslideProbability + '%';

            const segListEl = document.getElementById('segment-list');
            segListEl.innerHTML = route.segments.map(seg => `

                        <div class="flex space-x-3 text-xs text-slate-400 mt-1"><span>Rain: ${seg.rain}%</span><span>Slide: ${seg.landslide}%</span></div>
                    </div>
                    <span class="px-3 py-1 text-xs rounded-lg font-bold ${seg.landslide > 70 ? 'bg-red-950 text-red-300 border border-red-800' : 'bg-emerald-950 text-emerald-300 border border-emerald-800'}">${seg.status}</span>

                const poly = L.polyline(route.coordinates, { color: '#FFCC00', weight: 6 }).addTo(adminMap);
                adminMap.fitBounds(poly.getBounds(), { padding: [40, 40] });
            }
        }

        function speakText(text) {
            if (isAudioMuted) return;
            if ('speechSynthesis' in window) {

                utterance.rate = 1.0;
                window.speechSynthesis.speak(utterance);
            }
